from controller import Robot

# Инициализация робота и шаг симуляции
robot = Robot()
timestep = int(robot.getBasicTimeStep())

# Параметры управления
Kp = 0.003  # Коэффициент П-регулятора (подбирается экспериментально)
base_speed = 3.0  # Базовая скорость обоих колес

# Названия сенсоров освещенности у Khepera IV
sensor_names = [
    "ground left infrared sensor", 
    "ground front left infrared sensor",
    "ground front right infrared sensor",
    "ground right infrared sensor"
]

# Подключение сенсоров
sensors = []
for name in sensor_names:
    sensor = robot.getDevice(name)
    sensor.enable(timestep)
    sensors.append(sensor)

# Подключение моторов
left_motor = robot.getDevice("left wheel motor")
right_motor = robot.getDevice("right wheel motor")
left_motor.setPosition(float('inf'))
right_motor.setPosition(float('inf'))

# Основной цикл
while robot.step(timestep) != -1:
    # Чтение значений с сенсоров
    left_value = sensors[0].getValue()
    center_left_value = sensors[1].getValue()
    center_right_value = sensors[2].getValue()
    right_value = sensors[3].getValue()

    # Ошибка: белая линия имеет высокую яркость
    # Расчет ошибки как разности между левым и правым сенсором
    error = (right_value+center_right_value) - (left_value+center_left_value)

    # П-регулятор
    correction = Kp * error

    # Расчет скоростей колес
    left_speed = base_speed + correction
    right_speed = base_speed - correction

    # Ограничение скоростей (если нужно)
    max_speed = 6.28  # для Khepera IV
    left_speed = max(min(left_speed, max_speed), -max_speed)
    right_speed = max(min(right_speed, max_speed), -max_speed)

    # Установка скоростей
    left_motor.setVelocity(left_speed)
    right_motor.setVelocity(right_speed)
